# serasoft-migrate-1x.py Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `tools/serasoft-migrate-1x.py`, a tool that migrates a hop-process-template 1.0.x project to a given template tag and to the Serasoft Airflow platform layout, in six committed phases with a report, plus the export query and README documentation.

**Architecture:** One Python 3 file (standard library only) in `tools/`, not shipped to projects. Phase 0 analyses the project and builds a `Plan` without writing; phases 1-6 each change the checkout and commit on `migration/template-<tag>`. XML files are changed with targeted text replacements. pytest tests build three 1.0.x fixture projects from the template's own tags.

**Tech Stack:** Python 3.10+ (stdlib: argparse, csv, json, re, subprocess, xml.etree, dataclasses, pathlib), Git, `tools/serasoft-sync.sh` / `tools/serasoft-verify.sh`, pytest 8; Hop lab (`tests/hop`) for the final run.

**Spec:** `docs/superpowers/specs/2026-11-06-migrate-1x-design.md`

## Global Constraints

- Branch `feat/migrate-1x` (exists, holds the spec). One task = one commit. Never push until the controller opens the PR. English everywhere.
- **No real customer or project names** in code, tests, fixtures or docs: use `load_sales`, `proc_a`, `proc_b`, `sj_proc`, `myproject`, `sales.db.*`.
- Command: `tools/serasoft-migrate-1x.py --dest <checkout> --tag <X.Y.Z> --processes <csv> --env <name>=<file> [...] [--project <name>] [--db postgresql|mssql|sqlite]`. Exit codes: 0 complete, 2 complete with manual work, 1 stopped, 64 usage.
- Branch created on the project: `migration/template-<tag>`; commit messages exactly: `Migration: project structure for template <tag>`, `Migration: paths and variables for template <tag>`, `Migration: remove template 1.0.x`, `Migration: hop-process-template <tag>`, `Migration: configuration for the Serasoft Airflow platform`, `Migration: DB seed and report`.
- Manifest order: `config/template-env.json`, `config/template-<db>.json`, `config/<project>.json`, `config/<project>-<env>.json`, `/run/secrets/<project>_<env>`.
- Credentials: variable name matches `password|pwd|psw|secret|token` (case-insensitive) → `config/secrets-example.json` (empty value), never in other files.
- `serasoft.execution.log.filename` = `/tmp/<project>/logs/execution_log_${p_base_process_name}_${serasoft.project.main.workflow}_${v_process_run_timestamp}.log`; `serasoft.execution.file.name` = `/tmp/<project>/temp/execution_file_${p_base_process_name}_${serasoft.project.main.workflow}_${v_process_run_timestamp}.csv`.
- Tests need the local template repository to have tags `1.0.1` and an annotated `3.2.1` (both present); tests pass `--tag 3.2.1`.

~~~sql
-- Export of the template 1.0.x configuration, input of tools/serasoft-migrate-1x.py.
-- Run on the old integrations database (PostgreSQL):
--   psql "<connection string>" -v ON_ERROR_STOP=1 --csv -f tools/serasoft-migrate-1x-export.sql > export.csv
SELECT 'process' AS kind, CAST(id AS varchar) AS id, short_name, description, CAST(is_active AS varchar) AS is_active,
       CAST(NULL AS varchar) AS event_category_id, CAST(NULL AS varchar) AS event_type, CAST(NULL AS varchar) AS name
FROM config.integrations_processes
UNION ALL
SELECT 'category', CAST(id AS varchar), NULL, NULL, NULL, NULL, CAST(event_type AS varchar), name
FROM config.event_category
UNION ALL
SELECT 'subcategory', CAST(id AS varchar), NULL, NULL, NULL, CAST(event_category_id AS varchar), NULL, name
FROM config.event_subcategory
ORDER BY 1, 2;
~~~

def git(cwd, *args):
    return subprocess.run(["git", *args], cwd=cwd, check=True, capture_output=True, text=True,
                          env={**os.environ, **GIT_ENV}).stdout.strip()


def write(base, rel, text):
    path = Path(base) / rel
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text)
    return path


def _action(name, typ, x, y, extra=""):
    return (f"    <action>\n      <name>{name}</name>\n      <description/>\n      <type>{typ}</type>\n"
            f"      <attributes/>\n{extra}      <parallel>N</parallel>\n      <xloc>{x}</xloc>\n"
            f"      <yloc>{y}</yloc>\n      <attributes_hac/>\n    </action>\n")


START = _action("START", "SPECIAL", 100, 100,
                "      <DayOfMonth>1</DayOfMonth>\n      <hour>12</hour>\n      <intervalMinutes>60</intervalMinutes>\n"
                "      <intervalSeconds>0</intervalSeconds>\n      <minutes>0</minutes>\n      <repeat>N</repeat>\n"
                "      <schedulerType>0</schedulerType>\n      <weekDay>1</weekDay>\n")


def log_action(name, message, x=300):
    return _action(name, "WRITE_TO_LOG", x, 100,
                   f"      <loglevel>Basic</loglevel>\n      <logmessage>{message}</logmessage>\n      <logsubject>{name}</logsubject>\n")


def pipeline_action(name, filename, x=500):
    return _action(name, "PIPELINE", x, 100,
                   f"      <filename>{filename}</filename>\n      <loglevel>Basic</loglevel>\n      <parameters>\n"
                   f"        <pass_all_parameters>Y</pass_all_parameters>\n      </parameters>\n"
                   f"      <run_configuration>local</run_configuration>\n      <wait_until_finished>Y</wait_until_finished>\n")


def workflow(name, actions, chain):
    hops = "".join(f"    <hop>\n      <from>{a}</from>\n      <to>{b}</to>\n      <enabled>Y</enabled>\n"
                   f"      <evaluation>Y</evaluation>\n      <unconditional>{'Y' if a == 'START' else 'N'}</unconditional>\n    </hop>\n"
                   for a, b in zip(chain, chain[1:]))
    return (f'<?xml version="1.0" encoding="UTF-8"?>\n<workflow>\n  <name>{name}</name>\n'
            f"  <name_sync_with_filename>N</name_sync_with_filename>\n  <description/>\n  <extended_description/>\n"
            f"  <workflow_version/>\n  <parameters>\n  </parameters>\n  <actions>\n{START}{''.join(actions)}  </actions>\n"
            f"  <hops>\n{hops}  </hops>\n  <notepads>\n  </notepads>\n  <attributes/>\n</workflow>\n")


def pipeline(name, transforms, order):
    hops = "".join(f"    <hop>\n      <from>{a}</from>\n      <to>{b}</to>\n      <enabled>Y</enabled>\n    </hop>\n"
                   for a, b in zip(order, order[1:]))
    return (f'<?xml version="1.0" encoding="UTF-8"?>\n<pipeline>\n  <info>\n    <name>{name}</name>\n'
            f"    <name_sync_with_filename>Y</name_sync_with_filename>\n    <description/>\n    <extended_description/>\n"
            f"    <pipeline_version/>\n    <pipeline_type>Normal</pipeline_type>\n    <parameters>\n    </parameters>\n"
            f"  </info>\n  <notepads>\n  </notepads>\n  <order>\n{hops}  </order>\n{''.join(transforms)}"
            f"  <transform_error_handling>\n  </transform_error_handling>\n  <attributes/>\n</pipeline>\n")


def transform(name, typ, x, extra=""):
    return (f"  <transform>\n    <name>{name}</name>\n    <type>{typ}</type>\n    <description/>\n"
            f"    <distribute>Y</distribute>\n    <custom_distribution/>\n    <copies>1</copies>\n"
            f"    <partitioning>\n      <method>none</method>\n      <schema_name/>\n    </partitioning>\n{extra}"
            f"    <attributes/>\n    <GUI>\n      <xloc>{x}</xloc>\n      <yloc>100</yloc>\n    </GUI>\n  </transform>\n")


def dummy_pipeline(name):

def _from_tag(dest):
    dest.mkdir(parents=True)
    archive = subprocess.run(["git", "-C", str(ROOT), "archive", "1.0.1"], check=True, capture_output=True).stdout
    subprocess.run(["tar", "-x", "-C", str(dest)], input=archive, check=True)


def _commit(dest):
    git(dest, "init", "-q", "-b", "main")
    git(dest, "add", "-A")
    git(dest, "commit", "-q", "-m", "1.0.x project")


def root_workflow(dest, name, short_name, target, extra_params=None):
    """A copy of the 1.0.1 main.hwf for one process, saved as <name> at the project root."""
    text = subprocess.run(["git", "-C", str(ROOT), "show", "1.0.1:main.hwf"], check=True,
                          capture_output=True, text=True).stdout
    text = text.replace("<default_value>template</default_value>", f"<default_value>{short_name}</default_value>")
    text = text.replace("${PROJECT_HOME}/src/main/starter_job.hwf", target)
    for pname, default in (extra_params or {}).items():
        text = text.replace("  </parameters>",
                            f"    <parameter>\n      <name>{pname}</name>\n      <default_value>{default}</default_value>\n"
                            f"      <description/>\n    </parameter>\n  </parameters>", 1)
    write(dest, name, text)

def fail(message):
    print(f"ERROR: {message}", file=sys.stderr)

def run(cmd, cwd=None):
    r = subprocess.run(cmd, cwd=cwd, capture_output=True, text=True)
    if r.returncode != 0:
        fail(f"{' '.join(map(str, cmd))} failed: {r.stderr.strip() or r.stdout.strip()}")
    return r.stdout


def git(dest, *args):
    return run(["git", "-C", str(dest), *args]).strip()


def template_show(tag, path):

def template_files(tag):

def var_replace(text, mapping):
    """Replace ${old} and %%old%% references of every mapped variable; returns (text, count)."""
    count = 0
    for old in sorted(mapping, key=len, reverse=True):
        pattern = re.compile(r"(\$\{|%%)" + re.escape(old) + r"(\}|%%)")
        text, n = pattern.subn(lambda m: m.group(1) + mapping[old] + m.group(2), text)
        count += n
    return text, count


def rewrite_vars(text):
    return var_replace(text, {**TEMPLATE_VARS, **RUNTIME_VARS})


def classify(name):
    if name in DROPPED_VARS:
        return "drop", name
    mapped = TEMPLATE_VARS.get(name, name)
    if SECRET_RE.search(name):
        return "secret", mapped
    if name in TEMPLATE_VARS:
        return "template", mapped
    if name.startswith(("v_", "p_")):
        return "runtime", name
    return "project", name


def logic_signature(text):
    """Actions/transforms (type, SQL, referenced file name) and hops, independent of paths and layout."""
    root = ET.fromstring(text)
    sig = []
    for tag in ("action", "transform"):
        for el in root.iter(tag):
            fname = (el.findtext("filename") or "").strip()
            sig.append((tag, (el.findtext("name") or "").strip(), (el.findtext("type") or "").strip(),
                        " ".join((el.findtext("sql") or "").split()), PurePosixPath(fname).name))
    for hop in root.iter("hop"):
        sig.append(("hop", (hop.findtext("from") or "").strip(), (hop.findtext("to") or "").strip(),
                    (hop.findtext("enabled") or "").strip(), (hop.findtext("evaluation") or "").strip()))
    return sorted(sig)


def resolve_ref(var, rel, old_file):
    if var == "PROJECT_HOME":
        target = os.path.normpath(rel)
    else:
        target = os.path.normpath(os.path.join(os.path.dirname(old_file), rel))
    return None if target.startswith("..") else target


def find_root_workflows(dest, files):
    roots = []
    for path in sorted(f for f in files if "/" not in f and f.endswith(".hwf")):
        root = ET.parse(dest / path).getroot()
        targets = [(a.findtext("filename") or "").strip() for a in root.iter("action")
                   if (a.findtext("type") or "") == "WORKFLOW"]
        if not any(t.endswith("src/pre/main_pre_job.hwf") for t in targets):
            continue
        main = [t for t in targets if "/src/main/" in t]
        if len(main) != 1:
            fail(f"{path}: cannot find the process workflow (actions calling src/main/: {main})")
        m = REF_RE.search(main[0])
        target = resolve_ref(m.group(1), m.group(2), path) if m else None
        if not target or target not in files:
            fail(f"{path}: the process workflow {main[0]} does not exist")
        params, short_name, no_default = {}, "", []
        for p in root.iter("parameter"):
            name = (p.findtext("name") or "").strip()
            default = (p.findtext("default_value") or "").strip()
            if name == "p_base_process_name":
                short_name = default
            elif name not in ROOT_PARAMS:
                if default:
                    params[name] = default
                else:
                    no_default.append(name)
        roots.append(RootWorkflow(path, short_name, target, params, no_default))
    return roots


def read_csv(path):
    processes, categories, subcategories = {}, [], []
    with open(path, newline="") as fh:
        rows = list(csv.DictReader(fh))
    if not rows or "kind" not in rows[0]:
        fail(f"{path}: not an export of tools/serasoft-migrate-1x-export.sql")
    for r in rows:
        if r["kind"] == "process":
            processes[r["short_name"]] = r
        elif r["kind"] == "category":
            categories.append(r)
        elif r["kind"] == "subcategory":
            subcategories.append(r)
    return processes, categories, subcategories


def analyse(args):
    dest = Path(args.dest).resolve()
    if git(dest, "rev-parse", "--show-toplevel") != str(dest):
        fail(f"{dest} is not the root of a Git work tree")
    if git(dest, "status", "--porcelain"):
        fail(f"{dest} has uncommitted changes")
    branch = f"migration/template-{args.tag}"
    if subprocess.run(["git", "-C", str(dest), "rev-parse", "--verify", "--quiet", branch],
                      capture_output=True).returncode == 0:
        fail(f"branch {branch} already exists in {dest}")
    if run(["git", "-C", str(TEMPLATE_REPO), "cat-file", "-t", f"refs/tags/{args.tag}"]).strip() != "tag":
        fail(f"tag {args.tag} is not an annotated tag of {TEMPLATE_REPO}")
    files = set(git(dest, "ls-files").split("\n")) - {""}
    if "src/pre/main_pre_job.hwf" not in files:
        fail(f"{dest} is not a template 1.0.x project (no src/pre/main_pre_job.hwf)")
    report = Report()
    envs = {}
    for spec in args.env:
        name, _, path = spec.partition("=")
        p = Path(path).resolve()
        if not re.fullmatch(r"[A-Za-z0-9_.-]+", name) or not p.is_file():
            fail(f"--env {spec}: expected <name>=<existing file>")
        try:
            json.loads(p.read_text())["variables"]
        except (ValueError, KeyError):
            fail(f"--env {spec}: not a Hop environment file")
        envs[name] = p
    processes, categories, subcategories = read_csv(args.processes)

    roots = find_root_workflows(dest, files)
    if not roots:
        fail("no root workflow calling src/pre/main_pre_job.hwf")
    for r in roots:
        if not r.short_name:
            fail(f"{r.path}: empty p_base_process_name default")
        if r.short_name not in processes:
            fail(f"{r.path}: process {r.short_name} is not in {args.processes}")
        if r.params_without_default:
            report.params_without_default.append((r.short_name, r.params_without_default))
    report.unknown_processes = sorted(set(processes) - {r.short_name for r in roots})

    entries = {r.target for r in roots}
    template_present = [f for f in TEMPLATE_1X_FILES if f in files]
    removals = [f for f in template_present if f not in KEPT_FOR_LATER]
    removals += [r.path for r in roots if r.path not in removals]
    if "src/main/starter_job.hwf" in files and "src/main/starter_job.hwf" not in entries:
        removals.append("src/main/starter_job.hwf")
    for f in removals:
        if f.endswith((".hwf", ".hpl")) and f in TEMPLATE_1X_FILES:
            ref = template_show(REFERENCE_TAG, f)
            try:
                if ref and logic_signature(ref) != logic_signature((dest / f).read_text()):
                    report.customised.append(f)
            except ET.ParseError:
                report.customised.append(f)

    moves = {}
    def add_move(old, new):
        if new in moves.values() or new in files:
            fail(f"two sources for {new} (one of them {old}); check the processes {sorted(r.short_name for r in roots)}")
        moves[old] = new
    for r in roots:
        target_dir = f"src/project/main/{r.short_name}"
        old_dir = os.path.dirname(r.target)
        if old_dir == "src/main":
            add_move(r.target, f"{target_dir}/start.hwf")
            for f in sorted(files):
                if os.path.dirname(f) == "src/main" and f != r.target and f not in TEMPLATE_1X_FILES:
                    add_move(f, f"{target_dir}/{os.path.basename(f)}")
        else:
            for f in sorted(f for f in files if f.startswith(old_dir + "/")):
                new = f"{target_dir}/{f[len(old_dir) + 1:]}"
                if f == r.target:
                    new = f"{target_dir}/start.hwf"
                add_move(f, new)
    removed = set(removals)
    for f in sorted(files):
        if f in moves or f in removed or f in KEPT_FOR_LATER:
            continue
        if f.startswith("src/"):
            rest = f[len("src/"):]
            if rest.startswith("main/"):          # src/main/<x> not used by a root workflow
                rest = rest[len("main/"):]
            if rest.startswith("commons/"):       # src/commons/<x> and src/main/commons/<x>
                rest = rest[len("commons/"):]
            add_move(f, f"src/project/commons/{rest}")
        elif f.startswith("tools/"):
            add_move(f, f"src/project/{f}")

    tag_files = template_files(args.tag)
    equivalents = {f: "src/template/" + f[len("src/"):] for f in TEMPLATE_1X_FILES
                   if f.startswith("src/") and "src/template/" + f[len("src/"):] in tag_files}
    old_pc = None
    if "config/project-config.json" in files:
        old_pc = json.loads((dest / "config/project-config.json").read_text())
    for f in sorted(files):
        if os.path.basename(f) in DEPLOYMENT_FILES or f.endswith("hop-config.json"):
            report.deployment.append(f)
    project = args.project or dest.name
    if not re.fullmatch(r"[A-Za-z0-9_.-]+", project):
        fail(f"invalid project name {project}")
    return Plan(dest, args.tag, project, args.db, files, roots, processes, categories, subcategories,
                moves, removals, equivalents, old_pc, envs, report)


def parse_args(argv):
    p = argparse.ArgumentParser(description="Migrate a template 1.0.x project (see README.md).")
    p.add_argument("--dest", required=True)
    p.add_argument("--tag", required=True)
    p.add_argument("--processes", required=True)
    p.add_argument("--env", action="append", required=True, metavar="NAME=FILE")
    p.add_argument("--project")
    p.add_argument("--db", default="postgresql", choices=["postgresql", "mssql", "sqlite"])
    try:
        return p.parse_args(argv)
    except SystemExit as e:
        sys.exit(64 if e.code else 0)


def main(argv=None):
    args = parse_args(sys.argv[1:] if argv is None else argv)

    # Phases 1-6: added by the following tasks.
    print(f"analysis OK: {len(plan.roots)} process(es), {len(plan.moves)} move(s), {len(plan.removals)} removal(s)")
    return 0


if __name__ == "__main__":
    sys.exit(main())

def commit(plan, message):
    git(plan.dest, "add", "-A")
    git(plan.dest, "commit", "-q", "-m", message)


def phase_structure(plan):
    git(plan.dest, "checkout", "-q", "-b", f"migration/template-{plan.tag}")
    for old, new in sorted(plan.moves.items()):
        (plan.dest / new).parent.mkdir(parents=True, exist_ok=True)
        git(plan.dest, "mv", old, new)
        plan.report.moved.append((old, new))
    commit(plan, f"Migration: project structure for template {plan.tag}")


def rewrite_refs(plan, text, old_file):
    def repl(m):
        target = resolve_ref(m.group(1), m.group(2), old_file)
        if target is None:
            plan.report.unresolved.append((old_file, m.group(0)))
            return m.group(0)
        if target in plan.moves:
            new = plan.moves[target]
        elif target in plan.equivalents:
            new = plan.equivalents[target]
        elif target in plan.removals:
            plan.report.removed_refs.append((old_file, m.group(0)))
            return m.group(0)
        elif target in plan.files:
            new = target
        else:
            plan.report.unresolved.append((old_file, m.group(0)))
            return m.group(0)
        plan.report.rewrites["paths"] = plan.report.rewrites.get("paths", 0) + 1
        return "${PROJECT_HOME}/" + new
    return REF_RE.sub(repl, text)


def phase_rewrites(plan):
    reverse = {new: old for old, new in plan.moves.items()}
    current = set(git(plan.dest, "ls-files").split("\n")) - {""}
    removed = set(plan.removals)
    for path in sorted(current):
        old = reverse.get(path, path)
        if old in removed or old == "config/project-config.json":
            continue
        if not (path.endswith((".hwf", ".hpl")) or (path.startswith("metadata/") and path.endswith(".json"))):
            continue
        f = plan.dest / path
        text = f.read_text()
        new = rewrite_refs(plan, text, old)
        new, n = rewrite_vars(new)
        plan.report.rewrites["variables"] = plan.report.rewrites.get("variables", 0) + n
        new, n = re.subn(r"<connection>\s*integrations_db\s*</connection>",
                         "<connection>${serasoft.integrations.db.connection.name}</connection>", new)
        plan.report.rewrites["connections"] = plan.report.rewrites.get("connections", 0) + n
        if path.endswith(".hpl"):
            try:
                for t in ET.fromstring(new).iter("transform"):
                    if (t.findtext("table") or "").strip() == "integrations_log_events":
                        plan.report.events.append((path, (t.findtext("name") or "").strip()))
            except ET.ParseError:
                plan.report.unresolved.append((path, "not a valid XML file"))
        if new != text:
            f.write_text(new)
    commit(plan, f"Migration: paths and variables for template {plan.tag}")


def phase_removal(plan):
    present = [f for f in plan.removals if (plan.dest / f).exists()]

def phase_config(plan):
    d, p = plan.dest, plan.project
    defaults = {}
    for f in ("config/template-env.json", f"config/template-{plan.db}.json"):
        for v in json.loads((d / f).read_text())["variables"]:
            defaults[v["name"]] = v.get("value", "")
    secrets = []

    def classify_into(variables, origin):
        out = []
        for v in variables:
            name, value = v["name"], v.get("value", "") or ""
            kind, new = classify(name)
            value, _ = rewrite_vars(value)
            if kind == "drop":
                continue
            if kind == "runtime":
                plan.report.runtime_dropped.append((origin, name))
                continue
            if kind == "secret":
                if new not in secrets:
                    secrets.append(new)
                continue
            if kind == "template" and defaults.get(new) == value:
                continue
            if PATH_RE.match(value):
                plan.report.paths.append((origin, new, value))
            out.append(env_entry(new, value, f"From {name} ({origin})"))
        return out

    # project file and common variables
    template_pc = json.loads(template_show(plan.tag, "config/project-config.json"))
    old_vars = (plan.old_project_config or {}).get("config", {}).get("variables", [])
    write_json(d / "config/project-config.json", template_pc)
    write_json(d / f"config/{p}.json", {"variables": classify_into(old_vars, "config/project-config.json")})
    for r in plan.roots:
        if r.params:
            lines = [f"# Defaults of the extra parameters of {r.path} (template 1.0.x);",
                     "# override per execution with p_process_params (README.md, Process parameters)."]
            lines += [f"{k}={v}" for k, v in r.params.items()]
            (d / f"config/{r.short_name}.properties").write_text("\n".join(lines) + "\n")
    runtime = {
        "serasoft.execution.log.filename": f"/tmp/{p}/logs/execution_log_${{p_base_process_name}}_${{serasoft.project.main.workflow}}_${{v_process_run_timestamp}}.log",
        "serasoft.execution.file.name": f"/tmp/{p}/temp/execution_file_${{p_base_process_name}}_${{serasoft.project.main.workflow}}_${{v_process_run_timestamp}}.csv",
    }
    for env, path in sorted(plan.envs.items()):
        variables = classify_into(json.loads(path.read_text())["variables"], f"{env} environment")
        variables += [env_entry(k, v, "Writable path outside the read-only project folder") for k, v in runtime.items()]
        write_json(d / f"config/{p}-{env}.json", {"variables": variables})
        (d / "config/env").mkdir(parents=True, exist_ok=True)
        (d / f"config/env/{env}.list").write_text(
            f"# Hop environment files for {env}, in load order: a later file overrides an earlier one.\n"
            f"# Read by the Serasoft Airflow platform; cron: tools/serasoft-register-env.sh --env {env}\n"
            f"config/template-env.json\nconfig/template-{plan.db}.json\nconfig/{p}.json\nconfig/{p}-{env}.json\n"
            f"/run/secrets/{p}_{env}\n")
        rel = os.path.relpath(path, d)
        if not rel.startswith("..") and rel in set(git(d, "ls-files").split("\n")):
            git(d, "rm", "-q", "--cached", rel)
            path.unlink()
            plan.report.old_env_files_removed.append(rel)
    write_json(d / "config/secrets-example.json",
               {"variables": [env_entry(n, "", f"Credential: set it in the Docker secret {p}_<env>") for n in secrets]})
    plan.report.secrets = secrets
    gi = d / ".gitignore"
    lines = gi.read_text().splitlines() if gi.exists() else []
    lines = [l for l in lines if l.strip() not in ("config", "config/", "/config", "/config/")]
    lines += ["", "# Template 3.x: config/ is distributed and must be committed",
              "!config/", "!config/**", "config/*-local.json", "config/*secret*.json", "!config/secrets-example.json"]
    gi.write_text("\n".join(lines) + "\n")
    git(d, "add", "-A")
    git(d, "add", "-f", "config")

def phase_seed_report(plan):

        lines.append("INSERT INTO config.integrations_processes (id, short_name, description, project, is_active, is_running)\n"
                     f"SELECT {sql(r.short_name)}, {sql(r.short_name)}, {sql(row.get('description') or '')}, {sql(p)}, "
                     f"{sql((row.get('is_active') or 'Y')[:1])}, 'N'\n"
                     f"WHERE NOT EXISTS (SELECT 1 FROM config.integrations_processes WHERE short_name = {sql(r.short_name)});\n")
    for c in plan.categories:
